19def tb_version(): 

20 tb_version_proc = subprocess.Popen(["tensorboard", "--version"], stdout=subprocess.PIPE) 

21 proc_output = tb_version_proc.communicate() 

22 version_str = None 

23 for line in proc_output: 

24 if line is not None: 

25 line = str(line.decode("utf-8")).rstrip().lstrip() 

26 if "." in line: 

27 version_str = line 

28 

29 major, minor, *rest = version_str.split(".") 

30 return int(major), int(minor)
